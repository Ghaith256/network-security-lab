enable
configure terminal

hostname L3Switch-One
ip routing
file prompt quiet

! --- Uplink to Router-One (PRIMARY) via e1/Gi0/1 ---
interface GigabitEthernet0/1
 description LINK_TO_ROUTER_ONE_PRIMARY
 no switchport
 ip address 10.1.1.2 255.255.255.252
 no shutdown
exit

! --- Uplink to Router-Two (BACKUP) via e0/Gi0/0 ---
interface GigabitEthernet0/0
 description LINK_TO_ROUTER_TWO_BACKUP
 no switchport
 ip address 10.1.3.2 255.255.255.252
 no shutdown
exit

! --- Inter-Switch Link to L3Switch-Two via e2/Gi0/2 ---
interface GigabitEthernet0/2
 description LINK_TO_L3SWITCH_TWO
 no switchport
 ip address 10.1.5.1 255.255.255.252
 no shutdown
exit

! --- Trunk to Access Layer Switch via e3/Gi0/3 ---
interface GigabitEthernet0/3
 description TRUNK_TO_ACCESS_LAYER
 no shutdown
exit

interface GigabitEthernet0/3.10
 description IT_VLAN_GATEWAY
 encapsulation dot1Q 10
 ip address 192.168.20.1 255.255.255.0
 no shutdown
exit

interface GigabitEthernet0/3.20
 description HR_VLAN_GATEWAY
 encapsulation dot1Q 20
 ip address 192.168.21.1 255.255.255.0
 no shutdown
exit

! --- Routing to L3Switch-Two VLANs (Finance & Sales) ---
ip route 192.168.22.0 255.255.255.0 10.1.5.2
ip route 192.168.23.0 255.255.255.0 10.1.5.2

! --- Default Routes ---
ip route 0.0.0.0 0.0.0.0 10.1.1.1
ip route 0.0.0.0 0.0.0.0 10.1.3.1 250

end
write memory
